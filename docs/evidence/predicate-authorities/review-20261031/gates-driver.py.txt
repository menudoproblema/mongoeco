from pathlib import Path

def run(v):
 py=f'/tmp/mongoeco-predicate-c06/py{v}/bin/python';results={}
 commands={
 'install':[py,'-m','pip','install','--no-deps','--force-reinstall',str(wheel)],
 'pytest':[py,'-m','pytest','-q','-rs',*(['--cov=mongoeco','--cov-report=term','--cov-report=json:'+str(b/'coverage314.json')] if v=='314' else [])],
 'unittest':[py,'-m','unittest','discover','-s','tests','-p','test*.py'],

 }
 for name,command in commands.items():

 return v,results
