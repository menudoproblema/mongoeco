import json
import statistics
import time
from mongoeco import MongoClient
from mongoeco.engines import MemoryEngine, SQLiteEngine

out = {}
for factory in (MemoryEngine, SQLiteEngine):
    with MongoClient(factory()) as client:
        collection = client.bench.records
        collection.insert_many([{'_id': i, 'v': 0} for i in range(100)])
        collection.create_index([('v', 1)])
        samples = []
        checksums = []
        for sample in range(5):
            checksum = 0
            start = time.perf_counter()
            for i in range(300):
                value = (sample + i // 100 + 1) % 3
                result = collection.update_one({'_id': i % 100}, {'$set': {'v': value}})
                checksum += result.matched_count + result.modified_count
                checksum += collection.find_one({'_id': i % 100})['v']
            samples.append(time.perf_counter() - start)
            checksums.append(checksum)
        out[factory.__name__] = {'seconds': samples, 'median': statistics.median(samples), 'checksums': checksums}
for factory in (MemoryEngine, SQLiteEngine):
    with MongoClient(factory()) as client:
        collection = client.bench.records
        collection.insert_many([{'_id': i, 'v': 0} for i in range(100)])
        stream = collection.watch(max_await_time_ms=1)
        samples, checksums = [], []
        try:
            for sample in range(5):
                checksum = 0
                start = time.perf_counter()
                for i in range(300):
                    result = collection.update_one({'_id': i % 100}, {'$set': {'v': 0}})
                    checksum += result.matched_count + result.modified_count
                collection.insert_one({'_id': 'barrier-' + str(sample)})
                event = stream.try_next()
                assert event['operationType'] == 'insert'
                assert event['documentKey'] == {'_id': 'barrier-' + str(sample)}
                assert stream.try_next() is None
                samples.append(time.perf_counter() - start)
                checksums.append(checksum)
        finally:
            stream.close()
        out[factory.__name__ + '-watched-noop'] = {'seconds': samples, 'median': statistics.median(samples), 'checksums': checksums}
print(json.dumps(out, indent=2))
