class View:
 def __init__(self,c,entry):self.c,self.entry,self.returned=c,entry,None
 def __getattr__(self,n):return getattr(self.c,n)
 def update_one(self,q,u,**kwargs):
  if self.entry=='update-command':
   r=self.c.database.command({'update':self.c.name,'updates':[{'q':q,'u':u}]})
   if r.get('writeErrors'):
    e=r['writeErrors'][0];return SimpleNamespace(matched_count=r['n'],modified_count=r['nModified'],upserted_id=None,error=e)
   return SimpleNamespace(matched_count=r['n'],modified_count=r['nModified'],upserted_id=None)
  if self.entry=='find-command':
   r=self.c.database.command({'findAndModify':self.c.name,'query':q,'update':u,'new':True})
   self.returned=BSON.encode(r['value']).hex()
  elif isinstance(u,dict) and not any(k.startswith('$') for k in u):
   self.returned=BSON.encode(self.c.find_one_and_replace(q,u,return_document=True)).hex()
  else:self.returned=BSON.encode(self.c.find_one_and_update(q,u,return_document=True)).hex()
  return SimpleNamespace(matched_count=1,modified_count=None,upserted_id=None)
 def replace_one(self,*a,**k):return self.update_one(*a,**k)

def run(client):
 db=client['entry_'+uuid.uuid4().hex];out={}
 try:
  for entry in ('update-command','find-command','find-api'):
   for kind in ('classic','pipeline','replacement'):
    for pad in (0,1000):
     for noop in (False,True):
      seed={'_id':1,'v':1.0 if noop else 1,'pad':'x'*pad}
      update={'_id':1,'v':1.0,'pad':'x'*pad} if kind=='replacement' else [{'$set':{'v':1.0}}] if kind=='pipeline' else {'$set':{'v':1.0}}
      v=View(db.c,entry);r=_single(v,seed,update,mode='replacement' if kind=='replacement' else 'classic');r['returned']=v.returned;out[f'{entry}-{kind}-{pad}-{noop}']=r
  return out
 finally:client.drop_database(db.name)
