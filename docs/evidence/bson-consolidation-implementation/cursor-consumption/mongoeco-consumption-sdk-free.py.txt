def plain(value):
 if isinstance(value,list):return [plain(x) for x in value]
 if isinstance(value,dict):
  if len(value)==1 and ('$numberInt' in value or '$numberLong' in value):return int(next(iter(value.values())))
  return {key:plain(item) for key,item in value.items()}
 return value
def typed(value):
 if isinstance(value,dict):return ('document',tuple((key,typed(item)) for key,item in value.items()))
 if isinstance(value,list):return ('array',tuple(typed(item) for item in value))
 return (type(value).__name__,value)
