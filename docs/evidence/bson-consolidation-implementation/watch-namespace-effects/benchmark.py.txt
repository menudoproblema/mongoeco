import json,time,statistics
from mongoeco import MongoClient
from mongoeco.engines import MemoryEngine,SQLiteEngine
report={}
for backend,factory in [('memory',MemoryEngine),('sqlite',SQLiteEngine)]:
    for mode in ('default','updateLookup'):
        samples=[];checksums=[]
        for sample in range(5):
            with MongoClient(factory()) as client:
                c=client.benchmark.records
                c.insert_many([{'_id':i,'v':0,'pad':'x'*1000} for i in range(100)])
                s=c.watch(full_document=mode,max_await_time_ms=1)
                checksum=0
                try:
                    started=time.perf_counter()
                    for value in range(1,201):
                        c.update_one({'_id':1}, {'$set':{'v':value}})
                        event=s.try_next()
                        assert event['operationType']=='update'
                        assert event['documentKey']=={'_id':1}
                        if mode=='updateLookup':assert event['fullDocument']['v']==value
                        else:assert 'fullDocument' not in event
                        checksum+=value
                    samples.append(time.perf_counter()-started)
                    checksums.append(checksum)
                finally:s.close()
        report[backend+'-'+mode]={'samples':samples,'median':statistics.median(samples),'checksums':checksums}
print(json.dumps(report,indent=2))
