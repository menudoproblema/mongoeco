import asyncio,json,os,sys
from pathlib import Path
import pymongo
root=Path(Path('/tmp/mongoeco-dispatch-wheel-location.txt').read_text().strip())
site=Path(json.loads((root/'wheel.json').read_text())['site_packages']).resolve()
sys.path.insert(0,str(site))
import mongoeco
from mongoeco import MongoClient,AsyncMongoClient
from mongoeco.engines import MemoryEngine
assert Path(mongoeco.__file__).resolve().is_relative_to(site)
cases=[('empty',{}),('numeric-name',{1:1}),('scalar-name',5)]
def describe(error):
 return {'exception':type(error).__name__,'cause':None if error.__cause__ is None else type(error.__cause__).__name__,'code':getattr(error,'code',None)}
result={k:v for k,v in json.loads(Path('/tmp/mongoeco-bson-command-client-controls.json').read_text())['results'].items() if k.startswith('native/')}
for name,client in [('local',MongoClient(MemoryEngine()))]:
 with client:
  for label,spec in cases:
   try:client.test.command(spec)
   except Exception as error:result[f'{name}/sync/{label}']=describe(error)
   else:raise AssertionError(label)
async def exercise():
 for name,client in [('local',AsyncMongoClient(MemoryEngine()))]:
  async with client:
   for label,spec in cases:
    try:await client.test.command(spec)
    except Exception as error:result[f'{name}/async/{label}']=describe(error)
    else:raise AssertionError(label)
asyncio.run(exercise())
for surface in ('sync','async'):
 for label,_ in cases:assert result[f'native/{surface}/{label}']==result[f'local/{surface}/{label}'],result
Path('/tmp/mongoeco-bson-command-client-controls-final.json').write_text(json.dumps({'results':result,'site':str(site),'sdk':pymongo.version},indent=2)+'\n')
print('six paired client shape controls passed')
