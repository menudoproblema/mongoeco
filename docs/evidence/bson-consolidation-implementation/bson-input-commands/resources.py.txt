import json,subprocess
from pathlib import Path
subprocess.run(['docker','info','--format','{{.ServerVersion}}'],check=True,capture_output=True,text=True)
rows=[]
for round_id in range(1,7):
    root=Path('/tmp/mongoeco-bson-command-native'+('' if round_id==1 else f'-v{round_id}'))
    for target in ('7.0','8.0','9.0'):
        data=json.loads((root/target/'fixture.json').read_text())
        assert data['status']=='succeeded' and data['cleanup']['removed'] and data['cleanup']['volumes_removed']
        for kind,identifier in [('container',data['container_id']),*[('volume',v) for v in data['anonymous_volumes']]]:
            result=subprocess.run(['docker',kind,'inspect',identifier],capture_output=True,text=True)
            assert result.returncode and 'no such' in (result.stdout+result.stderr).lower(),(kind,identifier,result.stderr)
            rows.append({'round':round_id,'target':target,'kind':kind,'id':identifier,'absent':True})
for suffix in ('','-v2'):
    data=json.loads((Path('/tmp/mongoeco-bson-command-client-native'+suffix)/'fixture.json').read_text())
    assert data['status']==('failed' if not suffix else 'succeeded') and data['cleanup']['removed'] and data['cleanup']['volumes_removed']
    for kind,identifier in [('container',data['container_id']),*[('volume',v) for v in data['anonymous_volumes']]]:
        result=subprocess.run(['docker',kind,'inspect',identifier],capture_output=True,text=True)
        assert result.returncode and 'no such' in (result.stdout+result.stderr).lower()
        rows.append({'round':'client'+suffix,'target':'9.0','kind':kind,'id':identifier,'absent':True})
Path('/tmp/mongoeco-bson-command-resources.json').write_text(json.dumps(rows,indent=2)+'\n')
print('verified absent',len(rows),'resources')
