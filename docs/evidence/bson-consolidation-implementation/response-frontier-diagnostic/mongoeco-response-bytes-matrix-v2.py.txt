import os,subprocess,sys
from pathlib import Path
import yaml
root=Path('/tmp/mongoeco-response-bytes-probe-v2');root.mkdir(exist_ok=True)
payload=root/'capture.py'
payload.write_text(Path('/tmp/mongoeco-response-bytes-probe-v2.py').read_text())
for row in yaml.safe_load(Path('.github/workflows/mongodb-differential.yml').read_text())['jobs']['differential']['strategy']['matrix']['include']:
 v=row['target']
 with (root/(v+'.log')).open('w') as log:
  result=subprocess.run([sys.executable,'scripts/with_mongodb_replica_set.py','--image',row['image'],'--target',v,'--artifacts',str(root/v),'--timeout','600','--',sys.executable,str(payload)],env=dict(os.environ,MONGOECO_REAL_MONGODB_TARGET=v),stdout=log,stderr=subprocess.STDOUT)
 print(v,result.returncode,flush=True)
 if result.returncode:raise SystemExit(result.returncode)
