import json,subprocess
from pathlib import Path
subprocess.run(['docker','info','--format','{{.ServerVersion}}'],check=True,capture_output=True)
results=[]
for label in ('native','native-flags','native-final','native-frozen','native-consumers'):
 root=Path('/tmp/mongoeco-admin-read-'+label)
 for target in ('7.0','8.0','9.0'):
  fixture=json.loads((root/target/'fixture.json').read_text())
  assert fixture['status']=='succeeded' and fixture['cleanup']['removed'] and fixture['cleanup']['volumes_removed']
  resources=[('container',fixture['container_id'])]+[('volume',x) for x in fixture['anonymous_volumes']]
  for kind,name in resources:
   probe=subprocess.run(['docker',kind,'inspect',name],capture_output=True,text=True)
   assert probe.returncode and ('no such' in probe.stderr.lower() or 'not found' in probe.stderr.lower()),(kind,name,probe.stderr)
   results.append({'round':label,'target':target,'kind':kind,'name':name,'absent':True})
Path('/tmp/mongoeco-admin-read-resources.json').write_text(json.dumps(results,indent=2)+'\n')
print('absent resources',len(results))
