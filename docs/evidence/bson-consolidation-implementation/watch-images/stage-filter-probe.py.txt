import json,uuid
from pathlib import Path
from pymongo import MongoClient as Native
from mongoeco import MongoClient
from mongoeco.engines import MemoryEngine

def run(client):
    db=client['image_stage_'+uuid.uuid4().hex];out={}
    cases={
        'set_unrelated':[{'$set':{'tag':1}},{'$match':{'operationType':'insert'}}],
        'set_alias':[{'$set':{'kind':'$operationType'}},{'$match':{'kind':'insert'}}],
        'project_metadata':[{'$project':{'_id':1,'operationType':1,'fullDocument':1}},{'$match':{'operationType':'insert'}}],
        'unset_image':[{'$unset':'fullDocument'},{'$match':{'operationType':'insert'}}],
        'replace_merge':[{'$replaceWith':{'$mergeObjects':['$$ROOT',{'tag':1}]}},{'$match':{'operationType':'insert'}}],
        'set_operation_type':[{'$set':{'operationType':'insert'}},{'$match':{'operationType':'update'}}],
    }
    try:
        for name,pipeline in cases.items():
            c=db[name];c.insert_one({'_id':1,'v':0,'pad':'x'*1000})
            s=c.watch(pipeline,full_document='required',max_await_time_ms=20)
            try:
                c.update_one({}, {'$set':{'v':1}})
                try:out[name]={'empty':s.try_next() is None,'alive':s.alive}
                except Exception as e:out[name]={'error':type(e).__name__,'code':getattr(e,'code',None),'alive':s.alive}
            finally:s.close()
        return out
    finally:client.drop_database(db.name)
r={}
for v,p in [('7.0',49973),('8.0',49974),('9.0',49975)]:
    with Native(f'mongodb://127.0.0.1:{p}/?directConnection=true&serverSelectionTimeoutMS=3000') as c:r[v]=run(c)
with MongoClient(MemoryEngine()) as c:r['local']=run(c)
Path('docs/evidence/bson-consolidation-implementation/watch-images/stage-filter-probe.json').write_text(json.dumps(r,indent=2)+'\n')
print(r)
