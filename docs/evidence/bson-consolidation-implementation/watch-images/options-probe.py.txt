import json,uuid
from pathlib import Path
from pymongo import MongoClient
r={}
for version,port in [('7.0',49973),('8.0',49974),('9.0',49975)]:
    with MongoClient(f'mongodb://127.0.0.1:{port}/?directConnection=true&serverSelectionTimeoutMS=3000') as client:
        db=client['image_options_'+uuid.uuid4().hex]
        rows={}
        try:
            for i,value in enumerate([None,False,[],{}, {'wrong':True},{'enabled':False},{'enabled':True},{'enabled':0},{'enabled':1},{'enabled':'true'},{'enabled':None},{'enabled':True,'wrong':0}]):
                name='c'+str(i);db.create_collection(name)
                for command in ('create','collMod'):
                    target=name+'create' if command=='create' else name
                    try:
                        db.command({command:target,'changeStreamPreAndPostImages':value})
                        rows[command+str(i)]={'ok':True}
                    except Exception as e:rows[command+str(i)]={'error':type(e).__name__,'code':getattr(e,'code',None)}
            r[version]=rows
        finally:client.drop_database(db.name)
Path('docs/evidence/bson-consolidation-implementation/watch-images/options-probe.json').write_text(json.dumps(r,indent=2)+'\n')
print(r['9.0'])
