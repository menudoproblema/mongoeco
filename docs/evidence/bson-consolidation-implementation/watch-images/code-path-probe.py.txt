import json,uuid
from pathlib import Path
from bson import Code
from pymongo import MongoClient as Native
from mongoeco import MongoClient
from mongoeco.engines import MemoryEngine,SQLiteEngine

def run(client):
    db=client['code_path_'+uuid.uuid4().hex]
    try:
        c=db.c;c.insert_one({'_id':1,'operationType':'insert','n':1})
        out={}
        for name,expression in [('field',Code('$operationType')),('variable',Code('$$not_defined')),('literal',{'$literal':Code('$operationType')})]:
            try:
                row=list(c.aggregate([{'$project':{'_id':0,'v':expression}}]))[0]
                out[name]={'type':type(row['v']).__name__,'value':str(row['v'])}
            except Exception as e:out[name]={'error':type(e).__name__,'code':getattr(e,'code',None)}
        return out
    finally:client.drop_database(db.name)
r={}
for v,p in [('7.0',49973),('8.0',49974),('9.0',49975)]:
    with Native(f'mongodb://127.0.0.1:{p}/?directConnection=true&serverSelectionTimeoutMS=3000') as c:r[v]=run(c)
    for name,factory in [('memory',MemoryEngine),('sqlite',SQLiteEngine)]:
        with MongoClient(factory(),mongodb_dialect=v) as c:r[name+'-'+v]=run(c)
Path('docs/evidence/bson-consolidation-implementation/watch-images/code-path-probe.json').write_text(json.dumps(r,indent=2)+'\n')
print(r)
