import json,statistics,subprocess,sys
from pathlib import Path
base=Path(Path('/tmp/mongoeco-dispatch-baseline-location.txt').read_text().strip())/'src';root=Path(Path('/tmp/mongoeco-dispatch-wheel-location.txt').read_text().strip());candidate=Path(json.loads((root/'wheel.json').read_text())['site_packages'])
labels=('base','candidate','candidate','base')*2;rows=[]
for label in labels:
 result=subprocess.run([sys.executable,'/tmp/mongoeco-dispatch-benchmark.py',str(base if label=='base' else candidate)],capture_output=True,text=True,check=True)
 rows.append({'label':label,'cases':json.loads(result.stdout)})
ratios={}
for key in rows[0]['cases']:
 samples={label:[row['cases'][key] for row in rows if row['label']==label] for label in ('base','candidate')}
 assert len({x['bson_sha256'] for series in samples.values() for x in series})==1,key
 assert {x['count'] for series in samples.values() for x in series}=={200},key
 medians={label:statistics.median(x['cpu_ns'] for x in series) for label,series in samples.items()}
 ratios[key]={'base_median_ns':medians['base'],'candidate_median_ns':medians['candidate'],'ratio':medians['candidate']/medians['base']}
Path('/tmp/mongoeco-dispatch-cost.json').write_text(json.dumps({'order':labels,'samples':rows,'summary':ratios,'scope':'ABBA twice, own verification gates idle; diagnostic trend, not full D05 acceptance'},indent=2)+'\n')
print(json.dumps({k:round(v['ratio'],2) for k,v in ratios.items()}))
