import json,time,statistics
from bson import BSON
from mongoeco import MongoClient
from mongoeco.engines import MemoryEngine,SQLiteEngine
from mongoeco.core.bson_raw import decode_bson_document

out={'decode':{},'requests':{}}
for name,doc in [('ping',{'ping':1,'$db':'admin'}),('find',{'find':'records','filter':{'v':{'$gt':50}},'$db':'bench'}),('batch',{'insert':'records','documents':[{'_id':i,'v':i,'name':'value'} for i in range(20)],'$db':'bench'})]:
 raw=BSON.encode(doc)
 samples=[]
 for _ in range(5):
  start=time.perf_counter()
  for i in range(3000):
   result=decode_bson_document(raw)
  samples.append((time.perf_counter()-start)/3000)
 assert result==doc
 out['decode'][name]={'median_seconds_per_call':statistics.median(samples),'samples':samples}
for factory in (MemoryEngine,SQLiteEngine):
 with MongoClient(factory(),mongodb_dialect='7.0') as client:
  c=client.bench.records
  c.insert_many([{'_id':i,'v':i,'name':'value'} for i in range(100)])
  times=[]
  for _ in range(5):
   checksum=0
   start=time.perf_counter()
   for i in range(300):
    checksum+=sum(doc['v'] for doc in c.find({'v':{'$gt':50}}))
   times.append(time.perf_counter()-start)
  assert checksum==1102500
  out['requests'][factory.__name__]={'median_seconds':statistics.median(times),'samples':times,'checksum':checksum}
print(json.dumps(out,indent=2))
