import ast,hashlib,json
from pathlib import Path
paths=['src/mongoeco/api/_async/collection.py','src/mongoeco/api/_async/client.py','src/mongoeco/api/_async/database_admin.py']
rows=[]
for path in paths:
 source=Path(path).read_bytes();tree=ast.parse(source)
 for cls in (n for n in tree.body if isinstance(n,ast.ClassDef)):
  for node in cls.body:
   if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef)) and not node.name.startswith('_'):
    parameters=[a.arg for a in [*node.args.posonlyargs,*node.args.args,*node.args.kwonlyargs]]
    if node.name in {'list_indexes','index_information','drop_index','drop_indexes','drop','rename','list_collections','list_collection_names','validate_collection','create_search_index','create_search_indexes','update_search_index','drop_search_index','list_search_indexes'}:
     rows.append({'source':path,'source_sha256':hashlib.sha256(source).hexdigest(),'class':cls.name,'method':node.name,'line':node.lineno,'parameters':parameters,'kwargs':node.args.kwarg is not None,'disposition':'census for next B07 frontier audit, not a proven divergence or acceptance'})
Path('/tmp/mongoeco-bson-catalog-frontier-census.json').write_text(json.dumps(rows,indent=2)+'\n')
print(len(rows),'entrypoints require explicit disposition before closing B07; no new behavior claim')
