import json,os,statistics,subprocess,sys
from pathlib import Path
root=Path('docs/evidence/bson-consolidation-implementation/watch-token-admission')
runs={}
for label,source in [('before0','/tmp/mongoeco-token-base/src'),('after0','src'),('after1','src'),('before1','/tmp/mongoeco-token-base/src')]:
 env=dict(os.environ,PYTHONPATH=source)
 completed=subprocess.run([sys.executable,'/tmp/mongoeco-token-benchmark.py'],env=env,check=True,capture_output=True,text=True,timeout=120)
 data=json.loads(completed.stdout);runs[label]=data
 (root/f'benchmark-{label}.json').write_text(json.dumps(data,indent=2)+'\n')
 print('completed',label,flush=True)
summary={}
for engine in ['MemoryEngine','SQLiteEngine']:
 before=[value for label,data in runs.items() if label.startswith('before') for value in data[engine]['seconds']]
 after=[value for label,data in runs.items() if label.startswith('after') for value in data[engine]['seconds']]
 checksums={data[engine]['checksum'] for data in runs.values()}
 assert len(checksums)==1
 summary[engine]={'before_samples':len(before),'after_samples':len(after),'before_median_seconds':statistics.median(before),'after_median_seconds':statistics.median(after),'ratio':statistics.median(after)/statistics.median(before),'checksums_equal':True,'checksum_per_run':next(iter(checksums))}
(root/'benchmark-paired-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary,indent=2))
