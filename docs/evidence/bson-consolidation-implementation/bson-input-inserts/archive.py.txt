native=[]
for iteration in range(1,8):
 folder=Path('/tmp/mongoeco-bson-insert-native'+('' if iteration==1 else f'-v{iteration}'))
 for target in ('7.0','8.0','9.0'):
  fixture=json.loads((folder/target/'fixture.json').read_text());assert fixture['status']=='succeeded'
  (ROOT/f'native-{iteration}-{target}.json').write_text(json.dumps(fixture,indent=2)+'\n')
  (ROOT/f'native-{iteration}-{target}.txt.gz').write_bytes(gzip.compress((folder/(target+'.log')).read_bytes(),mtime=0))
  if iteration==7:
   reference=Path(f'tests/fixtures/semantic_bson_insert_{target.replace(".","_")}.json')
   check=folder/target/'bson_insert-check.json'
   assert reference.read_bytes()==check.read_bytes()
   shutil.copy2(check,ROOT/f'native-check-{target}.json')
   native.append({'target':target,'reference':str(reference),'sha256':hashlib.sha256(reference.read_bytes()).hexdigest(),'case_count':5,'observations':169,'check_byte_identical':True})

# All historical tracked fixtures and independent observers remain byte exact.
