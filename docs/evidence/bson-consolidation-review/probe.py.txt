def observe(fn):
    try:
        return {'ok': True, 'value': fn()}
    except Exception as exc:
        return {'ok': False, 'exception': type(exc).__name__, 'code': getattr(exc, 'code', None), 'message': str(exc)}


def run(client):
    db = client.bson_review
    coll = db.binary
    coll.insert_many([{'_id': 1, 'v': b'ab'}, {'_id': 2, 'v': Binary(b'ab', 0)}, {'_id': 3, 'v': Binary(b'ab', 5)}])
    filters = {
        'eq0': {'v': Binary(b'ab', 0)}, 'eq5': {'v': Binary(b'ab', 5)},
        'eq_bytes': {'v': b'ab'}, 'ne0': {'v': {'$ne': Binary(b'ab', 0)}},
        'gt0': {'v': {'$gt': Binary(b'a', 0)}},
        'expr_eq0': {'$expr': {'$eq': ['$v', Binary(b'ab', 0)]}},
    }
    r = {name: observe(lambda query=query: sorted(doc['_id'] for doc in coll.find(query))) for name, query in filters.items()}
    r['set_identical_binary'] = observe(lambda: coll.update_one({'_id': 3}, {'$set': {'v': Binary(b'ab', 5)}}).modified_count)
    coll = db.arrays
    oid = ObjectId('012345678901234567890123')
    coll.insert_many([{'_id': 1, 'v': [Binary(b'ab', 5)]}, {'_id': 2, 'v': [oid]}, {'_id': 3, 'v': [datetime(2000, 1, 1)]}])
    for alias in ['binData', 'objectId', 'date']:
        r['array_type_' + alias] = observe(lambda alias=alias: sorted(doc['_id'] for doc in coll.find({'v': {'$type': alias}})))
    coll = db.refs
    coll.insert_many([{'_id': 1, 'v': DBRef('c', 1)}, {'_id': 2, 'v': {'$ref': 'c', '$id': 1}}])
    for name, value in [('ref', DBRef('c', 1)), ('doc', {'$ref': 'c', '$id': 1})]:
        r['dbref_eq_' + name] = observe(lambda value=value: sorted(doc['_id'] for doc in coll.find({'v': {'$eq': value}})))
    r['group_dbref'] = observe(lambda: list(coll.aggregate([{'$group': {'_id': '$v', 'n': {'$sum': 1}}}])))
    coll = db.dates
    epoch = datetime(1970, 1, 1)
    coll.insert_many([{'_id': i, 'v': value} for i, value in enumerate([epoch + timedelta(milliseconds=1001), epoch - timedelta(milliseconds=1001), datetime(2038, 4, 22, 20, 55, 58, 687000)])])
    r['dates'] = observe(lambda: list(coll.aggregate([{'$project': {'_id': 1, 'long': {'$toLong': '$v'}, 'subtract': {'$subtract': ['$v', epoch]}, 'diff': {'$dateDiff': {'startDate': epoch, 'endDate': '$v', 'unit': 'millisecond'}}, 'roundtrip': {'$toDate': {'$toLong': '$v'}}}}])))
    for name, value in [('bad_expression', [{'$project': {'x': {'$notAnOperator': 1}}}]), ('replace_field_path', [{'$replaceWith': '$fullDocument'}])]:
        def opening(value=value):
            cursor = db.events.watch(value, max_await_time_ms=5)
            cursor.close()
            return 'opened'
        r['watch_open_' + name] = observe(opening)
    return r

native.drop_database('bson_review')  # Only the database owned by this disposable fixture.
